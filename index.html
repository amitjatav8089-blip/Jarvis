import speech_recognition as sr import pyttsx3 import webbrowser import datetime # Voice engine setup engine = pyttsx3.init() def speak(audio): engine.say(audio) engine.runAndWait() def take_command(): r = sr.Recognizer() with sr.Microphone() as source: print("Listening...") audio = r.listen(source) try: query = r.recognize_google(audio, language='en-in') return query.lower() except: return "" def main_logic(query): if 'arjun store' in query: speak("Arjun Store ki file access kar raha hoon, Amit.") # Yahan tum apni Excel file ka path daloge elif 'wikipedia' in query: speak("Searching Wikipedia...") # Yahan Wikipedia search logic aayega elif 'time' in query: strTime = datetime.datetime.now().strftime("%H:%M:%S") speak(f"Sir, the time is {strTime}") else: speak("Command samajh nahi aaya, please phir se kahein.") # Initial Run speak("Jarvis Online. Namaste Amit, main aapki kya madad kar sakta hoon?") while True: query = take_command() if 'exit' in query: break main_logic(query)
